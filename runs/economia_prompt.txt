```lua
--!strict
local HttpService = game:GetService("HttpService")
local Shared = game:GetService("ReplicatedStorage"):WaitForChild("Shared")

local CardData = require(Shared:WaitForChild("CardData"))
local PackData = require(Shared:WaitForChild("PackData"))
local TraitData = require(Shared:WaitForChild("TraitData"))
local CardGradeService = require(Shared:WaitForChild("CardGradeService"))
local PlayerDataStore = require(script.Parent.PlayerDataStore)

local CardService = {}

function CardService.RollCard(player: Player, packId: string?): (CardData.CardDef, TraitData.TraitDef)
	local data = PlayerDataStore.GetData(player)
	local totalLuck = (data.LuckMultiplier or 1.0)

	local pack = packId and PackData.GetPackById(packId)
	if pack then
		totalLuck = totalLuck * pack.LuckMultiplier
	end

	-- Weighted Roll for Card
	local weightedCards = {}
	local totalWeight = 0

	for _, card in ipairs(CardData.Cards) do
		local weight = card.Weight
		if card.Rarity ~= "Comum" then
			weight = weight * totalLuck
		end
		table.insert(weightedCards, { Card = card, Weight = weight })
		totalWeight += weight
	end

	local roll = math.random() * totalWeight
	local currentWeight = 0
	local selectedCard = CardData.Cards[1]

	for _, item in ipairs(weightedCards) do
		currentWeight += item.Weight
		if roll <= currentWeight then
			selectedCard = item.Card
			break
		end
	end

	-- Weighted Roll for Trait / Mutator
	local totalTraitWeight = 0
	local weightedTraits = {}
	for _, trait in pairs(TraitData.Traits) do
		local w = trait.Weight
		if trait.Id ~= "Normal" then
			w = w * (totalLuck ^ 0.5)
		end
		table.insert(weightedTraits, { Trait = trait, Weight = w })
		totalTraitWeight += w
	end

	local traitRoll = math.random() * totalTraitWeight
	local currentTraitWeight = 0
	local selectedTrait = TraitData.Traits.Normal

	for _, item in ipairs(weightedTraits) do
		currentTraitWeight += item.Weight
		if traitRoll <= currentTraitWeight then
			selectedTrait = item.Trait
			break
		end
	end

	return selectedCard, selectedTrait
end

function CardService.CreateCardInstance(cardDef: CardData.CardDef, traitDef: TraitData.TraitDef, level: number?)
	local lvl = level or 1
	local income = cardDef.BaseIncome * traitDef.Multiplier * (1 + (lvl - 1) * 0.25)
	local gradeInfo = CardGradeService.CalculateGrade(cardDef.Rarity, traitDef.Id, lvl)

	return {
		UniqueId = HttpService:GenerateGUID(false),
		Id = cardDef.Id,
		Name = cardDef.Name,
		Anime = cardDef.Anime,
		Rarity = cardDef.Rarity,
		Mutator = traitDef.Name,
		TraitId = traitDef.Id,
		MutatorMultiplier = traitDef.Multiplier,
		Level = lvl,
		Income = income,
		Grade = gradeInfo.Grade,
		GradeScore = gradeInfo.Score,
		GradeColor = { R = gradeInfo.Color.R, G = gradeInfo.Color.G, B = gradeInfo.Color.B },
	}
end

function CardService.UpgradeCardLevel(player: Player, invIndex: number): boolean
	local data = PlayerDataStore.GetData(player)
	local cardInst = data.Inventory[invIndex]
	if not cardInst then return false end

	local cost = math.floor(cardInst.Income * 15 * (cardInst.Level or 1))
	if data.Cash < cost then return false end

	data.Cash -= cost
	cardInst.Level = (cardInst.Level or 1) + 1
	
	local trait = TraitData.GetTrait(cardInst.TraitId or "Normal")
	local cardDef = CardData.GetCardById(cardInst.Id)
	local baseIncome = cardDef and cardDef.BaseIncome or 10
	
	cardInst.Income = baseIncome * trait.Multiplier * (1 + (cardInst.Level - 1) * 0.25)
	local newGrade = CardGradeService.CalculateGrade(cardInst.Rarity, trait.Id, cardInst.Level)
	cardInst.Grade = newGrade.Grade
	cardInst.GradeScore = newGrade.Score

	return true
end

function CardService.StartPassiveIncomeLoop()
	task.spawn(function()
		while true do
			task.wait(1)
			local Players = game:GetService("Players")
			for _, player in ipairs(Players:GetPlayers()) do
				local data = PlayerDataStore.GetData(player)
				if data and data.PlacedCards then
					local cashMultiplier = data.CashMultiplier or 1.0
					local totalPerSec = 0

					for _, cardInst in pairs(data.PlacedCards) do
						if cardInst and cardInst.Income then
							totalPerSec += cardInst.Income
						end
					end

					if totalPerSec > 0 then
						local earned = totalPerSec * cashMultiplier
						data.Cash = (data.Cash or 0) + earned
					end
				end
			end
		end
	end)
end

return CardService
```

### src/Shared/CardGradeService.luau
```lua
--!strict
local Shared = game:GetService("ReplicatedStorage"):WaitForChild("Shared")
local CardData = require(Shared:WaitForChild("CardData"))
local TraitData = require(Shared:WaitForChild("TraitData"))

export type GradeInfo = {
	Grade: string,
	Score: number,
	Color: Color3,
	Stars: number,
}

local CardGradeService = {}

local RARITY_POWER = {
	Comum = 10,
	Incomum = 30,
	Raro = 80,
	Épico = 200,
	Lendário = 500,
	Mítico = 1500,
	Divino = 5000,
	Caos = 20000,
}

function CardGradeService.CalculateGrade(rarity: string, traitId: string, level: number): GradeInfo
	local basePower = RARITY_POWER[rarity] or 10
	local trait = TraitData.GetTrait(traitId)
	local traitMult = trait and trait.Multiplier or 1.0
	local lvlMult = 1.0 + ((level or 1) - 1) * 0.25

	local score = basePower * traitMult * lvlMult

	local grade = "E"
	local color = Color3.fromRGB(150, 150, 150)
	local stars = 1

	if score >= 5000000 then
		grade = "SUPREMO+"
		color = Color3.fromRGB(255, 0, 255)
		stars = 7
	elseif score >= 1000000 then
		grade = "DIVINO"
		color = Color3.fromRGB(0, 255, 230)
		stars = 6
	elseif score >= 250000 then
		grade = "SSS"
		color = Color3.fromRGB(255, 215, 0)
		stars = 5
	elseif score >= 50000 then
		grade = "SS"
		color = Color3.fromRGB(255, 140, 0)
		stars = 5
	elseif score >= 10000 then
		grade = "S"
		color = Color3.fromRGB(160, 50, 240)
		stars = 4
	elseif score >= 2500 then
		grade = "A"
		color = Color3.fromRGB(50, 150, 250)
		stars = 3
	elseif score >= 800 then
		grade = "B"
		color = Color3.fromRGB(80, 200, 120)
		stars = 3
	elseif score >= 200 then
		grade = "C"
		color = Color3.fromRGB(200, 200, 200)
		stars = 2
	elseif score >= 50 then
		grade = "D"
		color = Color3.fromRGB(170, 170, 170)
		stars = 1
	end

	return {
		Grade = grade,
		Score = math.floor(score),
		Color = color,
		Stars = stars,
	}
end

return CardGradeService
```
